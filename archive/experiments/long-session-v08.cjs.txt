// One evolving game, rather than only fresh fixtures: import -> explore -> learn -> restore -> repeat.
const fs=require('node:fs'),assert=require('node:assert/strict'),crypto=require('node:crypto');
const started=Date.now(),deadline=Date.parse(JSON.parse(fs.readFileSync('research/runtime/v08_三小时续跑.json')).deadline),out='research/runtime/v08_长期存档流程.json';
const report={started:new Date(started).toISOString(),deadline:new Date(deadline).toISOString(),cycles:0,quests:0,batches:0,restores:0,physicalPickups:0,failures:[],versions:[],finished:false,environment:'One evolving Node game with simulated clock and JSON reload; not a browser session.'};let s=null,last='';
function write(){report.elapsedSeconds=Math.floor((Date.now()-started)/1000);if(s){report.uniqueQuestions=Object.values(s.progress).filter(p=>p.seen).length;report.inventory=s.play.inventory.length;report.storedTools=s.play.storage.length;report.combinations=s.play.combinations;report.bytes=Buffer.byteLength(JSON.stringify(s));}fs.writeFileSync(out+'.tmp',JSON.stringify(report,null,2));fs.renameSync(out+'.tmp',out);}
function cycle(){for(const name of ['venture-engine','learn-engine','learn-parser','material-compiler','world-catalog'])delete require.cache[require.resolve('../dist/'+name+'.js')];const L=require('../dist/learn-engine.js'),A=require('../dist/venture-engine.js'),P=require('../dist/learn-parser.js'),CAT=require('../dist/world-catalog.js');const sha=crypto.createHash('sha256').update(['learn-engine','venture-engine','material-compiler','learn-parser'].map(n=>fs.readFileSync('dist/'+n+'.js')).join('')).digest('hex');if(last!==sha){report.versions.push({sha,firstSeen:new Date().toISOString()});last=sha;}if(!s){s=L.fresh(808);const examples=require('../dist/sample-materials.js').cards,cards=Array.from({length:1000},(_,i)=>({...examples[i%examples.length],q:'【连续存档合成题 '+(i+1)+'】'+examples[i%examples.length].q}));const draft=P.parse(JSON.stringify({cards}),'连续存档.json'),bank=L.addBank(s,{...draft,title:'连续存档 · 1000道合成题'},started);s.banks=[bank];A.init(s).coupled=true;}s.zone=Object.keys(CAT.zones)[report.cycles%4];const now=started+report.cycles*60000;L.start(s,now);for(const o of CAT.objects(s.zone).filter(x=>!s.play.collected[x.id]).slice(0,2)){assert.equal(A.take(s,o.id,o.type).ok,true);report.physicalPickups++;}for(const q of A.session(s).quests){if(q.done)continue;const req=A.requirement(s,q);if(req&&!req.ready){let item=s.play.inventory.find(x=>x.type===req.tool);if(!item){const stored=s.play.storage.find(x=>x.type===req.tool);if(stored){if(s.play.inventory.length>=36)A.organize(s);assert.equal(A.retrieve(s,stored.id).ok,true);item=s.play.inventory.find(x=>x.id===stored.id);}else{const o=CAT.objects(s.zone).find(x=>x.type===req.tool&&!s.play.collected[x.id]);assert.ok(o,'physical prerequisite must remain reachable');assert.equal(A.take(s,o.id,o.type).ok,true);report.physicalPickups++;item=s.play.inventory.find(x=>x.id===o.id);}}assert.equal(A.combine(s,item.id,s.zone+'-'+req.kind,req.kind).ok,true);}for(const p of A.shuffle(q.pieces,report.cycles+q.id.length)){A.discover(s,q.id,p.key);if(!p.correct)assert.equal(A.place(s,q.id,p.key).ok,false);}for(const p of [...q.pieces].sort((a,b)=>(a.order||0)-(b.order||0))){assert.equal((p.correct?A.place(s,q.id,p.key,p.category):A.reject(s,q.id,p.key)).ok,true);}assert.equal(A.check(s,q.id,now).ok,true);report.quests++;}assert.ok(L.finish(s,now));report.batches++;s=L.restore(JSON.stringify(s));assert.ok(s);assert.equal(A.validate(s),true);report.restores++;L.start(s,now+60000,true);}
(async()=>{while(Date.now()<deadline){try{cycle();}catch(e){report.failures.push({at:new Date().toISOString(),cycle:report.cycles,message:e.stack});}report.cycles++;write();if(report.cycles%10===0)console.log(JSON.stringify({cycles:report.cycles,uniqueQuestions:report.uniqueQuestions,failures:report.failures.length}));await new Promise(r=>setTimeout(r,Math.min(30000,Math.max(1,deadline-Date.now()))));}report.finished=true;report.ended=new Date().toISOString();write();if(s)fs.writeFileSync('research/runtime/v08_长期存档最终快照.json',JSON.stringify(s));console.log(JSON.stringify(report));})();
